cmake_minimum_required(VERSION 3.20)
project(supervram LANGUAGES CXX)

option(SUPERVRAM_BUILD_TESTS "Build native tests" ON)

add_library(supervram_native STATIC src/supervram_native.cpp)
target_include_directories(supervram_native PUBLIC include)
target_compile_features(supervram_native PUBLIC cxx_std_17)
target_link_libraries(supervram_native PUBLIC ${CMAKE_DL_LIBS})

if(SUPERVRAM_BUILD_TESTS)
    enable_testing()
    add_executable(test_supervram_native tests/test_native.cpp)
    target_link_libraries(test_supervram_native PRIVATE supervram_native)
    add_test(NAME supervram_native COMMAND test_supervram_native)
endif()

# Optional: svram-verify correctness gate, linked against an existing llama.cpp build
# (set SUPERVRAM_LLAMA_BUILD to e.g. third_party/llama.cpp/build-3090)
set(SUPERVRAM_LLAMA_SRC "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp" CACHE PATH "llama.cpp source tree")
set(SUPERVRAM_LLAMA_BUILD "" CACHE PATH "llama.cpp build directory providing libllama/libggml")
option(SUPERVRAM_HAVE_EXPERT_CACHE "Compile svram-verify with compact GPU expert-cache support" OFF)
if(SUPERVRAM_LLAMA_BUILD AND EXISTS "${SUPERVRAM_LLAMA_BUILD}/bin/libllama.so")
    add_executable(svram-verify src/svram_verify.cpp)
    target_compile_features(svram-verify PRIVATE cxx_std_17)
    target_include_directories(svram-verify PRIVATE "${SUPERVRAM_LLAMA_SRC}/include" "${SUPERVRAM_LLAMA_SRC}/ggml/include")
    target_link_directories(svram-verify PRIVATE "${SUPERVRAM_LLAMA_BUILD}/bin")
    target_link_libraries(svram-verify PRIVATE llama ggml ggml-base)
    set_target_properties(svram-verify PROPERTIES BUILD_RPATH "${SUPERVRAM_LLAMA_BUILD}/bin")
    if(SUPERVRAM_HAVE_EXPERT_CACHE)
        target_compile_definitions(svram-verify PRIVATE SVRAM_HAVE_EXPERT_CACHE=1)
    endif()
    message(STATUS "SuperVRAM: building svram-verify against ${SUPERVRAM_LLAMA_BUILD}")
endif()

# Optional: real cuFile/GDS probe (built only when the CUDA toolkit and libcufile are available)
option(SUPERVRAM_BUILD_CUFILE_PROBE "Build the cuFile/GDS probe when CUDA is available" ON)
if(SUPERVRAM_BUILD_CUFILE_PROBE)
    find_package(CUDAToolkit QUIET)
    if(CUDAToolkit_FOUND AND TARGET CUDA::cuFile)
        add_executable(supervram_cufile_probe src/cufile_probe.cpp)
        target_compile_features(supervram_cufile_probe PRIVATE cxx_std_17)
        target_link_libraries(supervram_cufile_probe PRIVATE CUDA::cudart CUDA::cuFile)
        message(STATUS "SuperVRAM: building cuFile probe (CUDA ${CUDAToolkit_VERSION})")
    else()
        message(STATUS "SuperVRAM: cuFile probe skipped (CUDA toolkit or libcufile not found)")
    endif()
endif()
